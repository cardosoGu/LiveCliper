# V2 · Plano 2 — #painel, automático ligável e "tentar de novo" por plataforma

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** O admin controla tudo pelo `#painel` (liga/desliga o automático, escolhe categorias e mínimo de espectadores, liga plataformas, ajusta nota, ritmo e pausa, vê o status a cada 60 s) e pode reenviar para uma plataforma que falhou direto do card.

**Architecture:** `RuntimeSettings` ganha automático, categorias e espectadores. Um `AutoController` sobe e derruba, sob demanda, as tarefas do automático da V1 (chat, descoberta, clipper, clipes públicos) com as categorias resolvidas na Twitch. Um `Panel` independente do Discord monta o estado do painel e trata as ações; o `DiscordBot` só desenha a mensagem fixa (botões e selects) e repassa os cliques. O `Scheduler` ganha `publish_to` para reenviar a uma plataforma, e o `ApprovalDesk` mostra 🔁 nas que falharam.

**Tech Stack:** Python 3.12, discord.py 2.x, SQLite, pytest (asyncio automático).

## Global Constraints

- Spec: `docs/superpowers/specs/2026-11-17-v2-discord-painel-design.md` §4 (painel), §6 (automático), §8.3 (🔁 por plataforma).
- Canal do painel: `RuntimeSettings`/`DiscordBot.channels["painel"]` (ID em `DISCORD_PANEL_CHANNEL_ID`, já configurado).
- Chaves `kv` novas: `panel:auto_on` (`"1"`/`"0"`, padrão `"1"` para não mudar o comportamento de quem já roda a V1), `panel:categories` (JSON, lista de nomes), `panel:known_categories` (JSON, máx. 25), `panel:min_viewers`, `panel:message_id`.

- Desligado, o automático não roda nada (chat, descoberta, clipper e coletor de clipes públicos param). `pipeline`, `scheduler`, `approval`, `notifier` e `cleanup` seguem sempre ligados.
- Textos em português do Brasil.

```python
        self.db.kv_set("panel:categories", json.dumps(names, ensure_ascii=False))

        known = self.known_categories()

            return

            raise ValueError(f"o painel comporta no máximo {MAX_KNOWN_CATEGORIES} categorias")

        self.db.kv_set("panel:known_categories", json.dumps(known, ensure_ascii=False))

        return int(self.db.kv_get("panel:min_viewers", str(self.cfg.min_viewers)))

    def set_min_viewers(self, value: int) -> None:
        if value < 0:
            raise ValueError("o mínimo de espectadores não pode ser negativo")
        self.db.kv_set("panel:min_viewers", str(int(value)))
```

```python
"""Liga e desliga o automático (motor da V1: lives + picos no chat) conforme o painel."""
from __future__ import annotations

import asyncio
import logging
from typing import Awaitable, Callable

from ..core.supervisor import supervise
from .runtime import RuntimeSettings

log = logging.getLogger(__name__)

Jobs = dict[str, Callable[[], Awaitable[None]]]


class AutoController:
    def __init__(self, runtime: RuntimeSettings,
                 resolve: Callable[[list[str]], Awaitable[dict[str, str]]],
                 make_jobs: Callable[[list[str], int], Jobs],
                 alert: Callable[[str], Awaitable[None]],
                 on_stop: Callable[[], None] = lambda: None):
        self.runtime = runtime
        self.resolve = resolve
        self.make_jobs = make_jobs
        self.alert = alert
        self.on_stop = on_stop
        self.found: dict[str, str] = {}
        self.missing: list[str] = []
        self._tasks: list[asyncio.Task] = []
        self._lock = asyncio.Lock()

    @property
    def running(self) -> bool:
        return bool(self._tasks)

    async def stop(self) -> None:
        tasks, self._tasks = self._tasks, []
        for t in tasks:
            t.cancel()
        await asyncio.gather(*tasks, return_exceptions=True)
        if tasks:
            self.on_stop()

    async def apply(self) -> str:
        """Aplica o que está no painel: para tudo e, se ligado, sobe de novo."""
        async with self._lock:  # dois cliques seguidos não sobem tarefas em dobro
            await self.stop()
            if not self.runtime.auto_on():
                return "⚪ Automático desligado."
            wanted = self.runtime.categories()
            found = await self.resolve(wanted)
            self.found = found
            self.missing = [n for n in wanted if n.lower() not in {f.lower() for f in found}]
            if not found:
                return "⚠️ Nenhuma categoria encontrada na Twitch; o automático não foi ligado."
            jobs = self.make_jobs(list(found.values()), self.runtime.min_viewers())
            self._tasks = [asyncio.create_task(supervise(name, factory, self.alert))
                           for name, factory in jobs.items()]
            log.info("automático ligado: %s (mín. %d espectadores)",
                     ", ".join(found), self.runtime.min_viewers())
            msg = (f"🟢 Automático ligado · {', '.join(found)} · "
                   f"mín. {self.runtime.min_viewers()} espectadores")
            if self.missing:
                msg += f" · não encontradas na Twitch: {', '.join(self.missing)}"
            return msg

    def status_line(self) -> str:
        if not self.running:
            return "📡 Automático: DESLIGADO"
        return f"📡 Automático: LIGADO · {len(self.found)} categorias"
```

```python
"""#painel: estado e ações do painel de controle, independente do Discord."""
from __future__ import annotations

from dataclasses import dataclass
from typing import Awaitable, Callable

from .runtime import PLATFORMS, RuntimeSettings

VIEWER_OPTIONS = (300, 500, 1000, 2000, 5000, 10000)
SCORE_OPTIONS = (5, 6, 7, 8, 9)
INTERVAL_OPTIONS = (15, 30, 45, 60)
LIMIT_OPTIONS = (4, 8, 12, 20)
PLATFORM_LABEL = {"instagram": "Reels", "youtube": "Shorts", "tiktok": "TikTok"}


@dataclass
class PanelState:
    text: str
    auto_on: bool
    paused: bool
    known: list[str]
    selected: list[str]
    min_viewers: int
    min_score: float
    interval: int
    limit: int
    platforms: list[tuple[str, bool, bool]]  # (nome, ligada, configurada)


class Panel:
    def __init__(self, runtime: RuntimeSettings, auto, configured: list[str],
                 status_lines: Callable[[], list[str]],
                 resolve: Callable[[list[str]], Awaitable[dict[str, str]]]):
        self.runtime = runtime
        self.auto = auto
        self.configured = configured
        self.status_lines = status_lines
        self.resolve = resolve

    def state(self) -> PanelState:
        rt = self.runtime
        platforms = [(p, rt.platform_enabled(p), p in self.configured) for p in PLATFORMS]
        on = [PLATFORM_LABEL[p] for p, enabled, conf in platforms if enabled and conf]
        lines = [
            "## 🎛️ Painel Live Clips",
            self.auto.status_line(),
            f"Categorias: {', '.join(rt.categories())} · mín. {rt.min_viewers()} espectadores",
            f"Publicação: {'⏸️ pausada' if rt.paused() else '▶️ ativa'} · "
            f"plataformas: {', '.join(on) or 'nenhuma'} · nota mínima {rt.min_score():g} · "
            f"a cada {rt.post_interval_min()} min · até {rt.post_max_per_day()}/dia",
            *self.status_lines(),
        ]
        return PanelState(text="\n".join(lines), auto_on=rt.auto_on(), paused=rt.paused(),
                          known=rt.known_categories(), selected=rt.categories(),
                          min_viewers=rt.min_viewers(), min_score=rt.min_score(),
                          interval=rt.post_interval_min(), limit=rt.post_max_per_day(),
                          platforms=platforms)

    async def handle(self, action: str, values: list[str]) -> str:
        rt = self.runtime
        try:
            if action == "auto":
                rt.set_auto_on(not rt.auto_on())
                return await self.auto.apply()
            if action == "pause":
                rt.set_paused(not rt.paused())
                return "⏸️ Publicações pausadas." if rt.paused() else "▶️ Publicações retomadas."
            if action == "cats":
                rt.set_categories(values)
                return await self.auto.apply()
            if action == "viewers":
                rt.set_min_viewers(int(values[0]))
                return await self.auto.apply()
            if action == "score":
                rt.set_min_score(float(values[0]))
                return f"🎯 Nota mínima: {rt.min_score():g}"
            if action == "pace":
                interval, limit = values[0].split("|")
                rt.set_post_interval_min(int(interval))
                rt.set_post_max_per_day(int(limit))
                return f"⏱️ A cada {interval} min, até {limit} por dia."
            if action.startswith("plat-"):
                name = action.removeprefix("plat-")
                if name not in PLATFORMS:
                    return "Ação desconhecida."
                if name not in self.configured:
                    return f"{PLATFORM_LABEL[name]} não configurado no .env."
                rt.set_platform(name, not rt.platform_enabled(name))
                on = rt.platform_enabled(name)
                return f"{'🟢' if on else '⚪'} {PLATFORM_LABEL[name]} {'ligado' if on else 'desligado'}."
            if action == "addcat":
                name = values[0].strip() if values else ""
                found = await self.resolve([name]) if name else {}
                if not found:
                    return f"A categoria '{name}' não existe na Twitch."
                real = next(iter(found))
                rt.add_known_category(real)
                return f"➕ {real} adicionada à lista (marque no seletor para usar)."
        except (ValueError, IndexError) as e:
            return f"Valor inválido: {e}" if str(e) else "Valor inválido."
```
