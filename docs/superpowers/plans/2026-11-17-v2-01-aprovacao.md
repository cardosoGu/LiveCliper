# V2 · Plano 1 — Aprovação no Discord e publicação só do aprovado

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Todo clipe que a IA aprova cai num card no `#aprovação` (✅ Aprovar, ⚡ Agora, ✏️ Editar, ❌ Descartar) e o agendador só publica o que o admin aprovou, nas plataformas ligadas.

**Architecture:** Estados novos no `Candidate` (`awaiting_approval → approved → published`, ou `discarded`). Um `RuntimeSettings` (tabela `kv`) guarda o que o admin muda pelo Discord. Um `ApprovalDesk` independente do Discord (testável com transportes falsos) posta os cards e trata as ações; o `DiscordBot` só faz a ponte (canais, botões, modal). O `Scheduler` perde o modo ensaio e passa a ler intervalo, limite, pausa e plataformas do `RuntimeSettings`.

**Tech Stack:** Python 3.12, discord.py 2.x, SQLite, httpx, respx, pytest (asyncio automático).

## Global Constraints

- Spec: `docs/superpowers/specs/2026-11-17-v2-discord-painel-design.md` (§3, §8, §9, §11).
- Nomes dos canais: `painel`, `gerar-clipes`, `aprovação`, `logs`, dentro da categoria `Live Clips`; IDs salvos em `kv` como `discord_channel:<nome>`.
- O servidor do Discord é o do canal `DISCORD_CHANNEL_ID` que já existe; por isso **não** é preciso `DISCORD_GUILD_ID` (ajuste em relação ao §11 da spec). Esse canal só é usado para achar o servidor; os avisos passam para o `#logs`.
- Só `DISCORD_ADMIN_USER_ID` pode clicar nos botões; os outros recebem resposta efêmera.
- Nota mínima padrão passa a ser **6** (`AI_MIN_SCORE`).
- `DRY_RUN` deixa de existir como modo: só define o valor inicial das plataformas (`DRY_RUN=1` → todas começam desligadas).
- `TIKTOK_MODE` aceita `off | draft | private | public`; `draft` usa `/post/publish/inbox/video/init/`.
- Limite de upload do Discord: `DISCORD_UPLOAD_LIMIT_MB` (padrão 10).
- Textos para o usuário em português do Brasil, como no resto do código.
- Testes: `.venv/Scripts/python.exe -m pytest -q` (Windows) — a suíte inteira deve passar ao fim de cada tarefa.

"""Configurações que o admin muda pelo Discord, guardadas na tabela kv.

O .env só dá o valor inicial: depois que o admin muda algo, vale o que está no banco
(sobrevive a reinícios).
"""
from __future__ import annotations

from ..core.config import Settings
from ..core.db import DB

PLATFORMS = ("instagram", "youtube", "tiktok")


class RuntimeSettings:
    def __init__(self, db: DB, cfg: Settings):
        self.db = db
        self.cfg = cfg

    # nota mínima da IA (mesma chave que o /min_score da V1 usava)
    def min_score(self) -> float:
        return float(self.db.kv_get("min_score", str(self.cfg.ai_min_score)))

    def set_min_score(self, value: float) -> None:
        if not 0 <= value <= 10:
            raise ValueError("a nota precisa estar entre 0 e 10")
        self.db.kv_set("min_score", str(float(value)))

    # ritmo de publicação
    def post_interval_min(self) -> int:
        return int(self.db.kv_get("panel:post_interval_min",
                                  str(self.cfg.post_min_interval_min)))

    def set_post_interval_min(self, value: int) -> None:
        if value < 0:
            raise ValueError("o intervalo não pode ser negativo")
        self.db.kv_set("panel:post_interval_min", str(int(value)))

    def post_max_per_day(self) -> int:
        return int(self.db.kv_get("panel:post_max_per_day", str(self.cfg.post_max_per_day)))

    def set_post_max_per_day(self, value: int) -> None:
        if value < 1:
            raise ValueError("o limite precisa ser pelo menos 1")
        self.db.kv_set("panel:post_max_per_day", str(int(value)))

    # pausa geral das publicações (mesma chave do /pause da V1)
    def paused(self) -> bool:
        return self.db.kv_get("paused") == "1"

    def set_paused(self, on: bool) -> None:
        self.db.kv_set("paused", "1" if on else "0")

    # liga/desliga por plataforma (substitui o DRY_RUN)
    def platform_enabled(self, name: str) -> bool:
        default = "0" if self.cfg.dry_run else "1"
        return self.db.kv_get(f"panel:platform:{name}", default) == "1"

    def set_platform(self, name: str, on: bool) -> None:
        if name not in PLATFORMS:
            raise ValueError(f"plataforma desconhecida: {name}")
        self.db.kv_set(f"panel:platform:{name}", "1" if on else "0")
```

                 runtime: RuntimeSettings, now_fn: Callable[[], datetime] = utcnow,
                 render_fn: Callable[..., Awaitable[Path]] = render_final,

        self.cfg = cfg
        self.publishers = publishers
        self.media = media
        self.notifier = notifier
        self.runtime = runtime
        self.now_fn = now_fn
        self.render_fn = render_fn
        self.on_status = on_status
        self.fail_streak = 0
        self.retry_at: datetime | None = None
        self.attempts: dict[int, int] = {}

    def enabled_publishers(self) -> list:
        return [p for p in self.publishers if self.runtime.platform_enabled(p.name)]

    async def tick(self) -> bool:
        if self.runtime.paused() or not self.enabled_publishers():
            return False  # pausado ou tudo desligado: os aprovados esperam
        now = self.now_fn()
        if self.retry_at is not None and now < self.retry_at:
            return False
        approved = self.db.by_state(State.APPROVED, 200)
        if not approved:
            return False
        limit = self.runtime.post_max_per_day()
        posts_today = self.db.publications_count_since(local_midnight_utc(now))
        if posts_today >= limit:
            return False
        urgent = [c for c in approved if c.priority > 0]
        if urgent:  # "⚡ Agora": ignora o intervalo, não o limite diário
            c = min(urgent, key=lambda x: x.id)
        else:
            last = self.db.last_publication()
            if not can_post_now(now, last[0] if last else None, posts_today, limit,
                                self.runtime.post_interval_min()):
                return False
            c = min(approved, key=lambda x: x.id)  # ordem de aprovação
        await self.publish(c, now)
        return True
```

```python
"""#aprovação: cada clipe aprovado pela IA vira um card com botões; só o que o admin
aprova vai para a fila de publicação. Independente do Discord (o transporte é injetado)."""
from __future__ import annotations

import asyncio
import logging
import shutil
from pathlib import Path
from typing import Awaitable, Callable, Protocol

from ..ai.verdict import Verdict
from ..core.credits import credit_text
from ..core.db import DB
from ..core.models import Candidate, State
from ..core.timeutil import utcnow
from .runtime import RuntimeSettings

log = logging.getLogger(__name__)

ACTIONS = ("approve", "now", "edit", "discard")
AGE_WARN_H = 2  # cards do automático mostram a idade a partir daqui
PLATFORM_LABEL = {"instagram": "Reels", "youtube": "Shorts", "tiktok": "TikTok"}


def approval_custom_id(action: str, candidate_id: int) -> str:
    return f"appr:{action}:{candidate_id}"


def parse_custom_id(custom_id: str) -> tuple[str, int] | None:
    parts = custom_id.split(":")
    if (len(parts) != 3 or parts[0] != "appr" or parts[1] not in ACTIONS
            or not parts[2].isdigit()):
        return None
    return parts[1], int(parts[2])


class ApprovalTransport(Protocol):
    async def post(self, text: str, video: Path | None, link: str | None,
                   candidate_id: int, buttons: bool) -> int: ...

    async def edit(self, message_id: int, candidate_id: int, text: str,
                   buttons: bool) -> None: ...


class LogApprovalTransport:
    """Sem Discord configurado: os cards só aparecem no log."""

    async def post(self, text, video, link, candidate_id, buttons):
        log.info("[aprovação #%s] %s", candidate_id, text)
        return 0

    async def edit(self, message_id, candidate_id, text, buttons):
        log.info("[aprovação #%s] %s", candidate_id, text)


class ApprovalDesk:
    def __init__(self, db: DB, runtime: RuntimeSettings, transport: ApprovalTransport,
                 render: Callable[[Candidate, Verdict], Awaitable[bool]],
                 shrink: Callable[[Path, Path], Awaitable[Path]],
                 expose: Callable[[str], Awaitable[str]],
                 platform_names: list[str], upload_limit_mb: float = 10.0,
                 now_fn: Callable = utcnow):
        self.db = db
        self.runtime = runtime
        self.transport = transport
        self.render = render
        self.shrink = shrink
        self.expose = expose
        self.platform_names = platform_names
        self.limit_bytes = int(upload_limit_mb * 1024 * 1024)
        self.now_fn = now_fn
        self._age_shown: dict[int, int] = {}

    # --- cards ------------------------------------------------------------
    async def tick(self) -> int:
        posted = 0
        for c in self.db.by_state(State.AWAITING_APPROVAL, 50):
            if c.approval_message_id is None:
                if await self._post(c):
                    posted += 1
            else:
                await self._refresh_age(c)
        return posted

    async def _post(self, c: Candidate) -> bool:
        verdict = Verdict.model_validate_json(c.ai_json)
        if not await self.render(c, verdict):
            return False  # o render marcou o clipe como failed e avisou no #logs
        video, link = await self._attachment(Path(c.vertical_path))
        c.approval_message_id = await self.transport.post(
            self.card_text(c, verdict), video, link, c.id, True)
        self.db.update(c)
        self._age_shown[c.id] = self._age_hours(c)
        return True

    async def _attachment(self, path: Path) -> tuple[Path | None, str | None]:
        """Arquivo se couber no limite do Discord; senão uma versão leve; senão um link."""
        if path.stat().st_size <= self.limit_bytes:
            return path, None
        small = path.with_name("discord.mp4")
        try:
            await self.shrink(path, small)
            if small.stat().st_size <= self.limit_bytes:
                return small, None
        except Exception as e:
            log.warning("recompressão para o Discord falhou: %s", e)
        try:
            return None, await self.expose(str(path))
        except Exception as e:
            log.warning("link público falhou: %s", e)
            return None, None

    def _age_hours(self, c: Candidate) -> int:
        return int((self.now_fn() - c.created_at).total_seconds() // 3600)

    async def _refresh_age(self, c: Candidate) -> None:
        if c.source == "ondemand":
            return  # vídeo gravado não perde a graça com o tempo
        hours = self._age_hours(c)
        if hours < AGE_WARN_H or self._age_shown.get(c.id) == hours:
            return
        self._age_shown[c.id] = hours
        verdict = Verdict.model_validate_json(c.ai_json)
        await self.transport.edit(c.approval_message_id, c.id,
                                  self.card_text(c, verdict), True)

    def card_text(self, c: Candidate, verdict: Verdict) -> str:
        targets = ", ".join(PLATFORM_LABEL.get(n, n) for n in self.platform_names
                            if self.runtime.platform_enabled(n)) or "nenhuma (todas desligadas)"
        origin = f"📡 Automático · {credit_text(c.platform, c.channel)}"
        if c.source == "chat_spike":
            origin += " · pico no chat"
        lines = [f"🎬 **{verdict.title}** — nota {verdict.score:.1f}",
                 f"_{verdict.reason}_", origin, "",
                 verdict.caption, " ".join(verdict.hashtags), "",
                 f"Vai para: {targets}"]
        hours = self._age_hours(c)
        if c.source != "ondemand" and hours >= AGE_WARN_H:
            lines.append(f"⏰ há {hours} h")
        return "\n".join(lines)

    # --- ações ------------------------------------------------------------
    async def handle(self, action: str, candidate_id: int) -> str:
        c = self.db.get(candidate_id)
        if c is None:
            return "Clipe não encontrado."
        if c.state is not State.AWAITING_APPROVAL:
            return f"Esse clipe já foi decidido ({c.state.value})."
        if action in ("approve", "now"):
            c.state = State.APPROVED
            c.priority = 1 if action == "now" else 0
            c.reason = None  # daqui em diante, reason guarda falhas de publicação
            self.db.update(c)
            await self.on_status(c)
            return ("⚡ Aprovado: publica agora (respeitando o limite diário)."
                    if action == "now" else "✅ Aprovado: entrou na fila.")
        if action == "discard":
            c.state = State.DISCARDED
            c.reason = "descartado no Discord"
            self._delete_media(c)
            c.raw_path = c.vertical_path = None
            self.db.update(c)
            await self.on_status(c)
            return "❌ Descartado."
        return "Ação desconhecida."

    async def edit(self, candidate_id: int, title: str, caption: str) -> str:
        c = self.db.get(candidate_id)
        if c is None or c.state is not State.AWAITING_APPROVAL:
            return "Esse clipe não está mais aguardando aprovação."
        title, caption = title.strip(), caption.strip()
        if not title:
            return "O título não pode ficar vazio."
        verdict = Verdict.model_validate_json(c.ai_json).model_copy(
            update={"title": title, "caption": caption})
        c.ai_json = verdict.model_dump_json()
        if c.vertical_path and Path(c.vertical_path).name == "final.mp4":
            Path(c.vertical_path).unlink(missing_ok=True)  # o título está queimado no vídeo
        c.vertical_path = None
        old_message = c.approval_message_id
        c.approval_message_id = None  # o próximo tick refaz o vídeo e posta um card novo
        self.db.update(c)
        if old_message is not None:
            await self.transport.edit(old_message, c.id,
                                      "✏️ Editado — a nova versão chega em seguida.", False)
        return "✏️ Salvo. Refazendo o vídeo com o novo título."

    def _delete_media(self, c: Candidate) -> None:
        for p in (c.vertical_path, c.raw_path):
            if p and Path(p).parent.name == str(c.id):  # só a pasta do próprio clipe
                shutil.rmtree(Path(p).parent, ignore_errors=True)
                return

    # --- status depois da decisão -----------------------------------------
    async def on_status(self, c: Candidate) -> None:
        c = self.db.get(c.id) or c
        if not c.approval_message_id:
            return
        verdict = Verdict.model_validate_json(c.ai_json)
        text = self.card_text(c, verdict) + "\n\n" + self.status_line(c)
        await self.transport.edit(c.approval_message_id, c.id, text,
                                  c.state is State.AWAITING_APPROVAL)

    def status_line(self, c: Candidate) -> str:
        if c.state is State.APPROVED:
            line = ("⚡ aprovado — publicando agora" if c.priority
                    else "✅ aprovado — na fila de publicação")
            if c.reason:
                line += f" · última tentativa falhou: {c.reason}"
            return line
        if c.state is State.PUBLISHED:
            parts = []
            for p in self.db.publications_for(c.id):
                label = PLATFORM_LABEL.get(p["platform"], p["platform"])
                if p["url"]:
                    parts.append(f"{label} ✅ {p['url']}")
                elif p["platform"] == "tiktok":
                    parts.append(f"{label} 📥 rascunho/privado (abra o app do TikTok)")
                else:
                    parts.append(f"{label} 🔒 enviado (privado)")
            return "📤 publicado — " + " · ".join(parts)
        if c.state is State.DISCARDED:
            return "❌ descartado"
        if c.state is State.FAILED:
            return f"💥 falhou: {c.reason}"
        return ""

    async def run(self) -> None:
        while True:
            try:
                await self.tick()
            except asyncio.CancelledError:
                raise
            except Exception as e:
                log.exception("aprovação: %s", e)
            await asyncio.sleep(15)
```

    """Devolve {nome: canal}; cria o que faltar (inclusive se alguém apagou um canal)."""
    found: dict = {}

        if channel is None:
            if category is None:

                    or await guild.create_category(CATEGORY)
            channel = await guild.create_text_channel(name, category=category)
            db.kv_set(key, str(channel.id))
        found[name] = channel
    return found
